% Diagrams for the TT-SVD page. Each tikzpicture becomes one SVG.
%
% Build (from this folder):
%   latex diagrams.tex
%   dvisvgm --no-fonts --exact-bbox --page=1- --output=fig-%p diagrams.dvi
%   mv fig-1.svg ttsvd_reshape.svg
%   mv fig-2.svg ttsvd_svd_step.svg
%   mv fig-3.svg ttsvd_first_tensor.svg
%   mv fig-4.svg ttsvd_iteration.svg
%   mv fig-5.svg ttsvd_left_canonical.svg
%   mv fig-6.svg ttsvd_rounding.svg
%   rm -f diagrams.aux diagrams.log diagrams.dvi
%
% Colors and line weights follow the other figures on the site:
% gray = generic tensor, pink = isometry (U / left-orthogonal),
% blue circle = singular values, orange = V^dagger.

\documentclass[dvisvgm,tikz,border=4pt]{standalone}
\usepackage{newtxtext,newtxmath}
\usetikzlibrary{calc}

\definecolor{tgray}{HTML}{D5D5D5}
\definecolor{tpink}{HTML}{CB6D9D}
\definecolor{tblue}{HTML}{82A2DA}
\definecolor{torange}{HTML}{FFA902}

\tikzset{
  every picture/.style={line width=1.6pt, line cap=butt},
  sq/.style={draw, line width=1.6pt, minimum size=0.62cm, inner sep=0pt},
  gen/.style={sq, fill=tgray},
  iso/.style={sq, fill=tpink},
  vdag/.style={sq, fill=torange},
  sv/.style={draw, line width=1.6pt, circle, fill=tblue, minimum size=0.48cm, inner sep=0pt},
  pill/.style={draw, line width=1.6pt, fill=tgray, rounded corners=5pt,
               minimum height=0.62cm, inner sep=0pt},
  lbl/.style={font=\large},
  idx/.style={font=\normalsize},
  op/.style={font=\LARGE},
}

% \legs{node}{n}{width}: n evenly spaced legs on top of a pill of given width
\newcommand{\legs}[3]{%
  \foreach \j in {1,...,#2} {
    \draw ($(#1.north west)!{(\j-0.5)/#2}!(#1.north east)$) -- ++(0,0.5);
  }%
}

\begin{document}

% ------------------------------------------------------------------
% 1. Reshape the full tensor into a matrix
% ------------------------------------------------------------------
\begin{tikzpicture}
  \node[pill, minimum width=3.0cm] (T) at (0,0) {};
  \legs{T}{5}{3.0}
  \foreach \j/\s in {1/s_1, 2/s_2, 3/s_3, 4/\cdots, 5/s_N} {
    \node[idx, above] at ($(T.north west)!{(\j-0.5)/5}!(T.north east)+(0,0.5)$) {$\s$};
  }
  \node[lbl, below=3pt] at (T.south) {$T$};

  \node[op] at (2.5,0.15) {$\longrightarrow$};

  \node[gen, minimum size=0.8cm] (M) at (5.0,0.15) {};
  \draw (M.west) -- ++(-0.7,0) node[idx, left] {$s_1$};
  \draw (M.east) -- ++(0.7,0) node[idx, right] {$(s_2\, s_3 \cdots s_N)$};
  \node[lbl, below=3pt] at (M.south) {$M^{(1)}$};
\end{tikzpicture}

% ------------------------------------------------------------------
% 2. SVD of the first unfolding, with truncation
% ------------------------------------------------------------------
\begin{tikzpicture}
  \node[pill, minimum width=3.0cm] (T) at (0,0) {};
  \legs{T}{5}{3.0}
  \node[lbl, below=3pt] at (T.south) {$M^{(1)}$};

  \node[op] at (2.2,0) {$\approx$};

  \node[iso] (U) at (3.4,0) {};
  \draw (U.north) -- ++(0,0.5);
  \node[sv] (S) at (4.9,0) {};
  \node[pill, minimum width=2.4cm, fill=torange] (V) at (7.2,0) {};

  \draw (S.east) -- (V.west);
  \node[idx, above] at ($(U.east)!0.5!(S.west)$) {$\alpha_1$};

  \node[lbl, below=3pt] at (U.south) {$U^{(1)}$};
  \node[lbl, below=3pt] at (S.south |- U.south) {$S^{(1)}$};

% ------------------------------------------------------------------
% 3. First MPS/TT tensor and remainder
% ------------------------------------------------------------------
\begin{tikzpicture}

  \node[pill, minimum width=2.4cm] (R) at (2.4,0) {};

  \foreach \j/\s in {1/s_2, 2/s_3, 3/\cdots, 4/s_N} {

% ------------------------------------------------------------------
% 4. Full sequence of SVDs
% ------------------------------------------------------------------
\begin{tikzpicture}
  \def\dx{1.05}
  % row 1: full tensor
  \node[pill, minimum width=3.2cm] (T) at (1.6,0) {};
  \legs{T}{5}{3.2}

  % row 2
  \begin{scope}[yshift=-2.0cm]
    \node[iso] (a1) at (0,0) {};
    \node[pill, minimum width=2.4cm] (r) at (2.25,0) {};
    \legs{r}{4}{2.4}
  \end{scope}
  \draw (a1.north) -- ++(0,0.5);
  \draw (a1.east) -- (r.west);

  % row 3
  \begin{scope}[yshift=-4.0cm]
    \node[iso] (b1) at (0,0) {};
    \node[iso] (b2) at (\dx,0) {};
    \node[pill, minimum width=1.8cm] (rr) at (2.85,0) {};

  \end{scope}
  \foreach \n in {b1,b2} \draw (\n.north) -- ++(0,0.5);
  \draw (b1.east) -- (b2.west);
  \draw (b2.east) -- (rr.west);

  % row 4: final MPS/TT
  \begin{scope}[yshift=-6.6cm]
    \node[iso] (f1) at (0,0) {};
    \node[iso] (f2) at (\dx,0) {};
    \node[iso] (f3) at (2*\dx,0) {};
    \node[iso] (f4) at (3.9*\dx,0) {};
    \node[gen] (f5) at (4.9*\dx,0) {};
  \end{scope}
  \foreach \n in {f1,f2,f3,f4,f5} \draw (\n.north) -- ++(0,0.5);
  \draw (f1.east) -- (f2.west);
  \draw (f2.east) -- (f3.west);
  \draw (f3.east) -- ++(0.2,0);
  \draw (f4.west) -- ++(-0.2,0);
  \node at ($(f3.east)!0.5!(f4.west)$) {$\cdots$};

  \node[lbl, below=3pt] at (f2.south) {$A^{s_2}$};
  \node[lbl, below=3pt] at (f3.south) {$A^{s_3}$};
  \node[lbl, below=3pt] at (f4.south) {$A^{s_{N-1}}$};
  \node[lbl, below=3pt] at (f5.south) {$A^{s_N}$};

  % arrows
  \foreach \y/\t in {-0.55/SVD, -2.55/SVD} {
    \draw[->, line width=1.0pt] (-1.1,\y) -- ++(0,-0.95) node[midway, left, font=\large] {\t};
  }
  \node at (-1.1,-4.9) {$\vdots$};
  \draw[->, line width=1.0pt] (-1.1,-5.4) -- ++(0,-0.8) node[midway, left, font=\large] {SVD};
\end{tikzpicture}

% ------------------------------------------------------------------
% 6. TT-rounding of an existing MPS/TT
% ------------------------------------------------------------------
\begin{tikzpicture}
  \def\dx{1.05}
  % top: right-orthogonalized input (bond dimension R)
  \foreach \i in {1,...,5} {
    \ifnum\i=1
      \node[gen] (t\i) at (\i*\dx,0) {};
    \else
      \node[vdag] (t\i) at (\i*\dx,0) {};
    \fi
    \draw (t\i.north) -- ++(0,0.5);
  }
  \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,...,4}
    \draw[line width=3.2pt] (t\i.east) -- (t\j.west);
  \node[lbl, below=3pt] at ($(t1.south)!0.5!(t5.south)$) {right-orthogonal, bond dimension $R$};

  % bottom: after left-to-right SVD sweep (bond dimension r)
  \begin{scope}[yshift=-2.0cm]
    \foreach \i in {1,...,5} {
      \ifnum\i=5
        \node[gen] (u\i) at (\i*\dx,0) {};
      \else
        \node[iso] (u\i) at (\i*\dx,0) {};
      \fi
      \draw (u\i.north) -- ++(0,0.5);
    }
  \end{scope}
  \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,...,4}
    \draw (u\i.east) -- (u\j.west);
  \node[lbl, below=3pt] at ($(u1.south)!0.5!(u5.south)$) {left-orthogonal, bond dimension $r \leq R$};

  \draw[->, line width=1.0pt] (-0.1,-0.45) -- ++(0,-1.1)
     node[midway, left, font=\large, align=right] {SVD\\sweep};
\end{tikzpicture}
